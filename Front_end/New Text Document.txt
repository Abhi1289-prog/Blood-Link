 const [biometricConnected, setBiometricConnected] = useState(null);


const handleNext = () => {
    if (!biometricConnected) {
      alert('Biometric device is not connected!');
      return;
    }
    navigate('/blood-detection');
  };



 useEffect(() => {
    const detectBiometricDevice = async () => {
      if (!navigator.usb) {
        setBiometricConnected(false);
        return;
      }

      try {
        const devices = await navigator.usb.getDevices();
        setBiometricConnected(devices.length > 0);
      } catch (err) {
        console.error('USB Detection Error:', err);
        setBiometricConnected(false);
      }
    };

    detectBiometricDevice();
  }, []);





 {biometricConnected !== null && (
          <p className={`text-sm ${biometricConnected ? 'text-green-600' : 'text-red-500'}`}>
            {biometricConnected ? '✅ Biometric device is connected' : '❌ No biometric device found'}
          </p>
        )}
